import sys
from ncclient import manager
from ncclient.operations.rpc import RPCError
import xml.dom.minidom
import xml.etree.ElementTree as ET

print("--- Loading Libraries: ncclient, xml.dom.minidom, xml.etree.ElementTree ---")

# Replace this string with the actual required XML payload from the reference below
xml_payload = """<config>...</config>"""

dom = xml.dom.minidom.parseString(xml_payload)
print(f"--- XML PAYLOAD TO SEND ---\n{dom.toprettyxml(indent='  ')}")

try:
    print("--- STEP 2: Connecting to CSR1000v ---")
    with manager.connect(host="192.168.19.139", port=830, username="admin", password="123", 
                         hostkey_verify=False, device_params={'name': 'iosxe'}) as mgr:
        
        print("--- STEP 3: Sending edit-config request ---")
        response = mgr.edit_config(target='running', config=xml_payload)
        
        root = ET.fromstring(str(response))
        if root.find('.//{urn:ietf:params:xml:ns:netconf:base:1.0}ok') is not None:
            print("STATUS: Configuration applied successfully! (<ok/> received)")
        else:
            print("STATUS: Failed to apply configuration.")
            
        print("--- STEP 5: Verification (get-config) ---")
        verify_filter = """<filter><native xmlns="[http://cisco.com/ns/yang/Cisco-IOS-XE-native](http://cisco.com/ns/yang/Cisco-IOS-XE-native)">...</native></filter>"""
        v_response = mgr.get_config(source='running', filter=verify_filter)
        v_dom = xml.dom.minidom.parseString(str(v_response))
        print(f"--- LIVE ROUTER VERIFICATION CONFIG ---\n{v_dom.toprettyxml(indent='  ')}")

except RPCError as e:
    print(f"ERROR: NETCONF RPC Error encountered!\nTag: {e.tag}\nMessage: {e.message}")
except Exception as e:
    print(sys.stderr, f"ERROR: An unexpected exception occurred: {e}")
finally:
    print("\n--- FINAL SUMMARY - TASK COMPLETE ✓ ---")
    print("GitHub Repository: [https://github.com/Fedor-Goossens-pxl/lab-8-2-automation](https://github.com/Fedor-Goossens-pxl/lab-8-2-automation)")
